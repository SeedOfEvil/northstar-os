cmake_minimum_required(VERSION 3.28)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../../VERSION" WEB_DEMO_VERSION)
string(STRIP "${WEB_DEMO_VERSION}" WEB_DEMO_VERSION)
project(NorthstarWebDemo VERSION "${WEB_DEMO_VERSION}" LANGUAGES NONE)
set(WEB_DEMO_REVISION "development" CACHE STRING "Explicit sample source revision")
if(NOT WEB_DEMO_REVISION MATCHES "^[A-Za-z0-9._-]+$")
    message(FATAL_ERROR "WEB_DEMO_REVISION must be an identifier without whitespace or quoting")
endif()
configure_file(recipe.json.in recipe.json @ONLY)
configure_file(../../../LICENSE LICENSE COPYONLY)
configure_file(../NorthstarWelcome.app/Contents/Resources/northstar-welcome.svg icon.svg COPYONLY)
