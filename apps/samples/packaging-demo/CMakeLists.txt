cmake_minimum_required(VERSION 3.28)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../../VERSION" DEMO_VERSION)
string(STRIP "${DEMO_VERSION}" DEMO_VERSION)
project(NorthstarPackagingDemo VERSION "${DEMO_VERSION}" LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.6 REQUIRED COMPONENTS Gui Qml Quick)
add_executable(northstar-packaging-demo main.cpp)
target_link_libraries(northstar-packaging-demo PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
set(DEMO_REVISION "development" CACHE STRING "Source revision, supplied explicitly by the builder")
if(NOT DEMO_REVISION MATCHES "^[A-Za-z0-9._-]+$")
    message(FATAL_ERROR "DEMO_REVISION must be an identifier without whitespace or quoting")
endif()
configure_file(recipe.json.in recipe.json @ONLY)
configure_file(../../../LICENSE LICENSE COPYONLY)
configure_file(../NorthstarWelcome.app/Contents/Resources/northstar-welcome.svg icon.svg COPYONLY)
